public Verdict check(UUID playerId, Object packet) {
    State state = states.computeIfAbsent(playerId, id -> new State());
    long second = System.currentTimeMillis() / 1000L;

    synchronized (state) {
        if (second != state.windowSecond) {
            state.windowSecond = second;
            state.packetsInSecond = 0;
            state.packetsInSecondTotal = 0;
            state.perType.clear();
        }

        state.packetsInSecondTotal++;

        Integer typeLimit = limitFor(packetName);
        if (typeLimit != null) {
            int count = state.perType.merge(packetName, 1, Integer::sum);
            if (count > typeLimit) {
                flag(playerId, state, packetName + " over per-type limit ("
                        + count + "/" + typeLimit + ")");
                return observeMode ? Verdict.ALLOW : Verdict.DROP;
            }
        }

        if (maxPps > 0) {
            state.packetsInSecond++;
            if (state.packetsInSecond > maxPps) {
                if (dropOverLimit && !observeMode) {
                    return decideKick(state);
                }
            }
        }
    }
    return Verdict.ALLOW;
}
object MongoStorage {

    private val client = MongoClients.create(Settings.builder()
        .applyConnectionString(config("mongo.uri"))
        .build())

    private val database = client.getDatabase(config("mongo.database"))

    fun loadPlayer(uuid: UUID): PlayerData? =
        database.getCollection<PlayerData>("players")
            .find(Filters.eq("_id", uuid.toString()))
            .first()
            ?.let { decode<PlayerData>(it) }

    fun savePlayer(data: PlayerData) {
        database.getCollection<PlayerData>("players")
            .replaceOne(
                Filters.eq("_id", data.uuid.toString()),
                data,
                ReplaceOptions().upsert(true)
            )
    }
}
import subprocess
from pathlib import Path

def build(project: str, version: str) -> Path:
    """Compila el plugin y deja el JAR en target/."""
    root = Path("C:/repos") / project
    result = subprocess.run(
        ["mvn", "-q", "clean", "package"],
        cwd=root, capture_output=True, text=True
    )
    if result.returncode != 0:
        raise RuntimeError(result.stderr[-2000:])

    jars = sorted((root / "target").glob("*.jar"),
                   key=lambda p: p.stat().st_mtime)
    return jars[-1]


def deploy(jar: Path, server: str) -> None:
    """Sube el JAR al servidor y lo reinicia de forma segura."""
    plugins = Path(server) / "plugins"
    plugins.mkdir(parents=True, exist_ok=True)
    shutil.copy2(jar, plugins / jar.name)
    subprocess.run(["systemctl", "restart", server], check=True)

CREATE TABLE IF NOT EXISTS player_violations (
    player_uuid   UUID        NOT NULL,
    violation_lvl INTEGER     NOT NULL DEFAULT 0,
    packet_count  BIGINT      NOT NULL DEFAULT 0,
    last_alert_at TIMESTAMPTZ,
    PRIMARY KEY (player_uuid)
);

CREATE INDEX IF NOT EXISTS idx_violations_lvl
    ON player_violations (violation_lvl DESC);

-- Cache de datos en Redis + histórico de eventos en PostgreSQL
SELECT p.uuid, p.name, COALESCE(v.violation_lvl, 0) AS vl
FROM players p
LEFT JOIN player_violations v ON v.player_uuid = p.uuid
WHERE p.last_seen > NOW() - INTERVAL '30 days'
ORDER BY vl DESC
LIMIT 50;
