01 / perfil

Soy Francisco (@wbtt2), detrás de Centry Organizations. Lo que hago se reduce a dos cosas: dar mantenimiento a plugins y servidores Spigot/Paper que ya están en producción, y escribir plugins a medida para servidores y usuarios finales.

El día a día es corregir errores en caliente, adaptar plugins a versiones nuevas de Minecraft y refactorizar código que lleva años sin tocarse. Nada de forkear el servidor: uso reflexión y ProtocolLib para que el mismo JAR funcione en 1.8.8 y en 1.21.

Cuando un plugin no depende del servidor, escribo tooling en Python para builds y despliegues. Para Bedrock y herramientas .NET, C#. La infraestructura vive en Azure y AWS, con MongoDB para lo documental, Redis para caché y colas, y PostgreSQL/MySQL cuando el modelo es relacional.

object HubScoreboard {

    fun register(settings: HubConfig) {
        if (!settings.scoreboardEnabled) {
            Lumen.scoreboards().configuration = null
            return
        }

        Lumen.scoreboards().configuration = ScoreboardConfiguration()
            .title(TitleGetter.of(settings.scoreboardTitle))

    }

Lumen/src/main/kotlin/io/github/wbt/lumen/command/CommandManager.kt
package io.github.wbt.lumen.command

import io.github.wbt.lumen.command.CommandService
import io.github.wbt.lumen.command.LumenCommand
import io.github.wbt.lumen.utils.CC
import org.bukkit.Bukkit
import org.bukkit.command.Command
import org.bukkit.command.CommandMap
import org.bukkit.command.CommandSender
import org.bukkit.command.SimpleCommandMap
import org.bukkit.plugin.Plugin
import java.lang.reflect.Field
import java.util.concurrent.ConcurrentHashMap

internal class CommandManager : CommandService {
    private val registered = ConcurrentHashMap<String, Command>()

    private val commandMap: CommandMap by lazy {
        val server = Bukkit.getServer()
        val field: Field = server.javaClass.getDeclaredField("commandMap")
        field.isAccessible = true
        field.get(server) as CommandMap
    }

    override fun register(plugin: Plugin, command: LumenCommand) {
        val bukkitCommand = object : Command(command.name, command.description, command.usage, command.aliases) {
            override fun execute(sender: CommandSender, label: String, args: Array<out String>): Boolean {
                if (command.permission != null && !sender.hasPermission(command.permission)) {
                    sender.sendMessage(CC.RED + "No permission.")
                    return true
                }
                command.execute(sender, args.toList())
                return true
            }

            override fun tabComplete(sender: CommandSender, alias: String, args: Array<out String>): MutableList<String> {
                return command.complete(sender, args.toList()).toMutableList()
            }
        }
        commandMap.register(plugin.name.lowercase(), bukkitCommand)
        registered[command.name.lowercase()] = bukkitCommand
    }

    override fun unregister(name: String) {
        val command = registered.remove(name.lowercase()) ?: return
        command.unregister(commandMap)
        if (commandMap is SimpleCommandMap) {
            knownCommands()?.entries?.removeIf { it.value === command }
        }
    }

    fun unregisterAll() {
        registered.keys.toList().forEach { unregister(it) }
    }

    @Suppress("UNCHECKED_CAST")
    private fun knownCommands(): MutableMap<String, Command>? {
        return try {
            val field = SimpleCommandMap::class.java.getDeclaredField("knownCommands")
            field.isAccessible = true
            field.get(commandMap) as MutableMap<String, Command>
        } catch (ex: Exception) {
            null
        }
    }
}
Morgan/commons/src/main/kotlin/io/github/wbt/morgan/cache/RedisCache.kt
package io.github.wbt.morgan.cache

import com.google.gson.reflect.TypeToken
import io.github.wbt.morgan.Morgan
import io.github.wbt.morgan.redis.RedisPacketManager
import java.lang.reflect.Type
import java.util.concurrent.CompletableFuture

abstract class RedisCache<A, B>(
    var redisKey: String
) {

    var aToBCache: MutableMap<A, B> = mutableMapOf()
    var btoACache: MutableMap<B, A> = mutableMapOf()

    fun loadAllFromRedis() : CompletableFuture<Void>
    {
        return CompletableFuture.runAsync {
            val type: Type = object : TypeToken<MutableMap<A, B>>() {}.type
            val otherType: Type = object : TypeToken<MutableMap<B, A>>() {}.type
            val start = System.currentTimeMillis()

            RedisPacketManager.pool.resource.use {
                val ex = it.exists("$redisKey:A")

                //a -> b cache
                if (ex)
                {
                    val all = it.get("$redisKey:A")
                    val gson = Morgan.gson.fromJson<MutableMap<A, B>>(all, type)

                    this.aToBCache = gson
                }

                val ex2 = it.exists("$redisKey:B")

                //b -> a cache
                if (ex2)
                {
                    val all = it.get("$redisKey:B")
                    val gson = Morgan.gson.fromJson<MutableMap<B, A>>(all, otherType)

                    this.btoACache = gson
                }

                println("Cache load for $redisKey loaded in ${System.currentTimeMillis().minus(start)}ms")
            }

        }
    }

    fun addToFirstCache(key: A, value: B) : CompletableFuture<B>
    {
        return CompletableFuture.supplyAsync {
            aToBCache[key] = value

            RedisPacketManager.pool.resource.use {
                it.set(
                    "$redisKey:A",
                    Morgan.gson.toJson(aToBCache)
                )
            }
            value
        }
    }
}
Nobody/src/main/java/io/github/wbt/nobody/compat/menu/Menu.java
package io.github.wbt.nobody.compat.menu;

import org.bukkit.Bukkit;
import org.bukkit.ChatColor;
import org.bukkit.entity.Player;
import org.bukkit.event.inventory.ClickType;
import org.bukkit.inventory.Inventory;
import org.bukkit.inventory.InventoryHolder;

import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public abstract class Menu {

    public boolean isUpdateAfterClick = true;

    private final Map<UUID, Map<Integer, Button>> buttons = new HashMap<>();

    public abstract int size(Player player);

    public abstract String getTitle(Player player);

    public abstract Map<Integer, Button> getButtons(Player player);

    public final void openMenu(Player player) {
        final Map<Integer, Button> buttons = this.getButtons(player);

        this.buttons.put(player.getUniqueId(), buttons);

        final Inventory inventory = Bukkit.createInventory(new MenuHolder(this), this.size(player), color(this.getTitle(player)));

        for (final Map.Entry<Integer, Button> entry : buttons.entrySet()) {
            if (entry.getKey() != null && entry.getKey() >= 0 && entry.getKey() < inventory.getSize()) {
                inventory.setItem(entry.getKey(), entry.getValue().getButtonItem(player));
            }
        }

        currentlyOpened.put(player.getUniqueId(), this);

        player.openInventory(inventory);
    }

    public final void handleClick(Player player, int slot, ClickType clickType) {
        final Map<Integer, Button> buttons = this.buttons.get(player.getUniqueId());

        if (buttons == null) {
            return;
        }

        final Button button = buttons.get(slot);

        if (button == null) {
            return;
        }

        button.clicked(player, slot, clickType);

        if (this.isUpdateAfterClick && currentlyOpened.get(player.getUniqueId()) == this) {
            final Inventory inventory = player.getOpenInventory().getTopInventory();

            if (inventory.getHolder() instanceof MenuHolder && ((MenuHolder) inventory.getHolder()).menu == this) {
                for (final Map.Entry<Integer, Button> entry : buttons.entrySet()) {
                    if (entry.getKey() != null && entry.getKey() >= 0 && entry.getKey() < inventory.getSize()) {
                        inventory.setItem(entry.getKey(), entry.getValue().getButtonItem(player));
                    }
                }

                player.updateInventory();
            }
        }
    }

    public final void handleClose(Player player) {
        if (currentlyOpened.get(player.getUniqueId()) == this) {
            currentlyOpened.remove(player.getUniqueId());
        }

        this.buttons.remove(player.getUniqueId());
    }

    private static final Map<UUID, Menu> currentlyOpened = new HashMap<>();

    public static Map<UUID, Menu> getCurrentlyOpenedMenus() {
        return currentlyOpened;
    }

    public static String color(String text) {
        return ChatColor.translateAlternateColorCodes('&', text);
    }
}

public final class PacketLimiter {

    public enum Verdict {ALLOW, DROP, KICK}

    // Legacy (pre-1.17) packet names -> modern normalized name, so old config keys keep working.
    private static final Map<String, String> LEGACY_ALIASES = Map.ofEntries(
            Map.entry("windowclick", "containerclick"),
            Map.entry("setcreativeslot", "setcreativemodeslot"),
            Map.entry("blockplace", "useitemon"),
            Map.entry("usesitem", "useitem"),
            Map.entry("blockdig", "playeraction"),
            Map.entry("flying", "moveplayer"),
            Map.entry("look", "moveplayer"),
            Map.entry("position", "moveplayer"),
            Map.entry("positionlook", "moveplayer"),
            Map.entry("armanimation", "swing"),
            Map.entry("helditemslot", "setheldslot"),
            Map.entry("entityaction", "playercommand")
    );

    public Verdict check(UUID playerId, Object packet) {
        if (maxPps <= 0 && packetLimits.isEmpty()) {
            return Verdict.ALLOW;
        }

        String packetName = PacketNames.of(packet);
        State state = states.computeIfAbsent(playerId, id -> new State());
        long second = System.currentTimeMillis() / 1000L;

        synchronized (state) {
            if (second != state.windowSecond) {
                state.windowSecond = second;
                state.packetsInSecond = 0;
                state.packetsInSecondTotal = 0;
                state.perType.clear();
            }

            state.packetsInSecondTotal++;

            Integer typeLimit = limitFor(packetName);
            if (typeLimit != null) {
                int count = state.perType.merge(packetName, 1, Integer::sum);
                if (count > typeLimit) {
                    flag(playerId, state, packetName + " over per-type limit (" + count + "/" + typeLimit + ")");
                    return observeMode ? Verdict.ALLOW : Verdict.DROP;
                }
            }

            if (maxPps > 0) {
                state.packetsInSecond++;
                if (state.packetsInSecond > maxPps) {
                    if (state.packetsInSecond == maxPps + 1 || state.packetsInSecond % 50 == 0) {
                        flag(playerId, state, "PPS " + state.packetsInSecondTotal + "/" + maxPps);
                    }
                    if (dropOverLimit && !observeMode) {
                        return decideKick(state);
                    }
                }
            }
        }
        return Verdict.ALLOW;
    }

    private void decay() {
        long now = System.currentTimeMillis();
        Iterator<Map.Entry<UUID, State>> iterator = states.entrySet().iterator();
        while (iterator.hasNext()) {
            Map.Entry<UUID, State> entry = iterator.next();
            State state = entry.getValue();
            synchronized (state) {
                state.vl = Math.max(0, state.vl - decayPerInterval);
                if (state.vl == 0) {
                    state.alertStep = 0;
                }
            }
            boolean idle = now - (state.windowSecond * 1000L) > 60_000L;
            if (state.vl == 0 && (idle || plugin.getServer().getPlayer(entry.getKey()) == null)) {
                iterator.remove();
            }
        }
    }
}
